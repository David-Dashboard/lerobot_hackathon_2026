depthai               # OAK-D-PRO (Luxonis) — scene camera over DepthAI

qualia-sdk            # cloud ACT/VLA finetuning (Qualia)
python-dotenv         # load HF_TOKEN / QUALIA_TOKEN from .env
pyyaml                # config.yaml (arm serials, camera sizes, workspace/IK/ROI)
opencv-python         # webcam capture + homography (usually pulled in by lerobot too)
transformers>=4.57.1,<5   # zero-shot perception (OWL-ViT). <5 keeps huggingface-hub <1.0
